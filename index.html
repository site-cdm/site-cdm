function ajouterHistorique(utilisateurConnecte) { var historique = JSON.parse(localStorage.getItem('journalConnexions')) || []; var nouvelleConnexion = { utilisateur: utilisateurConnecte, date: new Date().toLocaleString('fr-FR') }; historique.push(nouvelleConnexion); localStorage.setItem('journalConnexions', JSON.stringify(historique)); }
